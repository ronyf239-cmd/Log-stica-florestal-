// =============================== // CÁLCULO DE PRODUÇÃO - HARVESTER // =============================== // CONFIGURAÇÕES const PRODUTIVIDADE_BASE = 1.8; // ha por máquina const HORAS_OPERACIONAIS = 19.5; const HORAS_CORRIDAS_DIA = 24; /** * Calcula o prazo de produção * * @param {number} area - Área em hectares * @param {number} maquinas - Quantidade de máquinas * @param {string} horarioInicio - Horário da medição, exemplo: "10:00" * @returns {object} resultado do cálculo */ function calcularProducao(area, maquinas, horarioInicio) { // Validação if (area <= 0 || maquinas <= 0) { throw new Error("Área e quantidade de máquinas devem ser maiores que zero."); } // Produção total das máquinas em 19,5 horas operacionais const producaoPorCiclo = PRODUTIVIDADE_BASE * maquinas; // Quantidade de ciclos necessários const ciclos = area / producaoPorCiclo; // Horas operacionais necessárias const horasOperacionais = ciclos * HORAS_OPERACIONAIS; // Converter horas operacionais para horas corridas const horasCorridas = ciclos * HORAS_CORRIDAS_DIA; // Calcular previsão de conclusão const [hora, minuto] = horarioInicio.split(":").map(Number); const inicio = new Date(); inicio.setHours(hora, minuto, 0, 0); const conclusao = new Date( inicio.getTime() + horasCorridas * 60 * 60 * 1000 ); return { area, maquinas, produtividadePorMaquina: PRODUTIVIDADE_BASE, producaoPorCiclo, ciclos, horasOperacionais, horasCorridas, horarioInicio: inicio, horarioConclusao: conclusao }; } // =============================== // EXEMPLO // =============================== const resultado = calcularProducao( 3.6, // hectares 2, // máquinas "10:00" // horário da medição ); console.log("Área:", resultado.area, "ha"); console.log("Máquinas:", resultado.maquinas); console.log( "Produção:", resultado.producaoPorCiclo.toFixed(2), "ha / 19,5 h operacionais" ); console.log( "Horas operacionais:", resultado.horasOperacionais.toFixed(1) ); console.log( "Horas corridas:", resultado.horasCorridas.toFixed(1) ); console.log( "Conclusão:", resultado.horarioConclusao.toLocaleTimeString("pt-BR", { hour: "2-digit", minute: "2-digit" }) );
